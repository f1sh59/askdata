## 3. 澄清后仍然失败的题

### v04 · 正确（仅列顺序不同）

- 原问题：各渠道的数据怎么样？
- 用户补充：2025 年，我要每个渠道的销售额和有效订单数，按渠道名分组。

```sql
-- gold
SELECT c.channel_name, COUNT(*) AS order_cnt, ROUND(SUM(o.pay_amount), 2) AS gmv FROM orders o JOIN channels c ON c.channel_id = o.channel_id WHERE o.status IN ('已支付','已发货','已完成') AND o.order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01' GROUP BY 1 ORDER BY gmv DESC

-- generated
SELECT
    c.channel_name,
    SUM(o.pay_amount) AS sales_amount,
    COUNT(DISTINCT o.order_id) AS valid_order_count
FROM orders o
JOIN channels c ON o.channel_id = c.channel_id
WHERE o.status IN ('已支付', '已发货', '已完成')
  AND o.order_time >= TIMESTAMP '2025-01-01'
  AND o.order_time < TIMESTAMP '2026-01-01'
GROUP BY c.channel_name
ORDER BY sales_amount DESC
```

### v07 · 结果值不一致

- 原问题：各会员等级表现如何？
- 用户补充：2025 年，每个会员等级的销售额和订单数，等级缺失的算「未知」。

```sql
-- gold
SELECT COALESCE(u.member_level, '未知'), COUNT(*) AS order_cnt, ROUND(SUM(o.pay_amount), 2) AS amt FROM orders o JOIN users u ON u.user_id = o.user_id WHERE o.status IN ('已支付','已发货','已完成') AND o.order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01' GROUP BY 1 ORDER BY amt DESC

-- generated
SELECT
    COALESCE(u.member_level, '未知') AS member_level,
    SUM(oi.item_amount) AS sales_amount,
    COUNT(DISTINCT o.order_id) AS order_count
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
LEFT JOIN users u ON o.user_id = u.user_id
WHERE o.status IN ('已支付', '已发货', '已完成')
  AND o.order_time >= TIMESTAMP '2025-01-01'
  AND o.order_time < TIMESTAMP '2026-01-01'
GROUP BY COALESCE(u.member_level, '未知')
ORDER BY sales_amount DESC
```

### v08 · 列数不一致

- 原问题：同比表现怎么样？
- 用户补充：看销售额，2025 年对比 2024 年的同比增长率，给百分比。

```sql
-- gold
WITH a AS (SELECT SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付','已发货','已完成') AND order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01'), b AS (SELECT SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付','已发货','已完成') AND order_time >= TIMESTAMP '2024-01-01' AND order_time < TIMESTAMP '2025-01-01') SELECT ROUND((a.amt - b.amt) / b.amt * 100, 2) FROM a, b

-- generated
WITH yearly_sales AS (
    SELECT
        EXTRACT(YEAR FROM o.order_time) AS year,
        SUM(oi.item_amount) AS sales_amount
    FROM orders o
    JOIN order_items oi ON o.order_id = oi.order_id
    WHERE o.status IN ('已支付', '已发货', '已完成')
      AND o.order_time >= TIMESTAMP '2024-01-01'
      AND o.order_time < TIMESTAMP '2026-01-01'
    GROUP BY EXTRACT(YEAR FROM o.order_time)
)
SELECT
    y2025.sales_amount AS sales_2025,
    y2024.sales_amount AS sales_2024,
    ROUND((y2025.sales_amount - y2024.sales_amount) / y2024.sales_amount * 100, 2) AS yoy_growth_rate_pct
FROM yearly_sales y2025
CROSS JOIN yearly_sales y2024
WHERE y2025.year = 2025 AND y2024.year = 2024
```

### v13 · 列数不一致

- 原问题：今年做得好吗？
- 用户补充：看 2025 年销售额，跟 2024 年比增长了多少百分比。

```sql
-- gold
WITH a AS (SELECT SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付','已发货','已完成') AND order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01'), b AS (SELECT SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付','已发货','已完成') AND order_time >= TIMESTAMP '2024-01-01' AND order_time < TIMESTAMP '2025-01-01') SELECT ROUND((a.amt - b.amt) / b.amt * 100, 2) FROM a, b

-- generated
WITH yearly_sales AS (
    SELECT
        CASE
            WHEN order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01' THEN '2025'
            WHEN order_time >= TIMESTAMP '2024-01-01' AND order_time < TIMESTAMP '2025-01-01' THEN '2024'
        END AS year,
        SUM(pay_amount) AS sales_amount
    FROM orders
    WHERE status IN ('已支付', '已发货', '已完成')
      AND order_time >= TIMESTAMP '2024-01-01'
      AND order_time < TIMESTAMP '2026-01-01'
    GROUP BY 1
)
SELECT
    MAX(CASE WHEN year = '2025' THEN sales_amount END) AS sales_2025,
    MAX(CASE WHEN year = '2024' THEN sales_amount END) AS sales_2024,
    (MAX(CASE WHEN year = '2025' THEN sales_amount END) - MAX(CASE WHEN year = '2024' THEN sales_amount END))
        / NULLIF(MAX(CASE WHEN year = '2024' THEN sales_amount END), 0) * 100 AS growth_rate_pct
FROM yearly_sales
```

### v14 · 结果值不一致

- 原问题：最近的退款和销售对比
- 用户补充：2025 年全年的退款金额、销售额和退款率，一行三个数。

```sql
-- gold
WITH r AS (SELECT SUM(refund_amount) AS amt FROM refunds WHERE status = '已完成' AND refund_time >= TIMESTAMP '2025-01-01' AND refund_time < TIMESTAMP '2026-01-01'), s AS (SELECT SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付','已发货','已完成') AND order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01') SELECT ROUND(s.amt, 2), ROUND(r.amt, 2), ROUND(r.amt / s.amt * 100, 2) FROM s, r

-- generated
SELECT
    (SELECT COALESCE(SUM(refund_amount), 0)
     FROM refunds
     WHERE status = '已完成'
       AND refund_time >= TIMESTAMP '2025-01-01'
       AND refund_time < TIMESTAMP '2026-01-01') AS refund_amount,
    (SELECT COALESCE(SUM(pay_amount), 0)
     FROM orders
     WHERE status IN ('已支付', '已发货', '已完成')
       AND order_time >= TIMESTAMP '2025-01-01'
       AND order_time < TIMESTAMP '2026-01-01') AS sales_amount,
    (SELECT COALESCE(SUM(refund_amount), 0)
     FROM refunds
     WHERE status = '已完成'
       AND refund_time >= TIMESTAMP '2025-01-01'
       AND refund_time < TIMESTAMP '2026-01-01')
    /
    NULLIF((SELECT COALESCE(SUM(pay_amount), 0)
            FROM orders
            WHERE status IN ('已支付', '已发货', '已完成')
              AND order_time >= TIMESTAMP '2025-01-01'
              AND order_time < TIMESTAMP '2026-01-01'), 0) AS refund_rate
```
