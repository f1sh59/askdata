### m10 · medium · 列数不一致

- 问题：2025 年每个月的退款率是多少？（退款率 = 已完成退款金额 / 销售额 × 100）
- 口径：退款额按退款时间归月，不是按订单时间；无退款的月份补 0

```sql
-- gold
WITH gmv AS (SELECT strftime(order_time, '%Y-%m') AS ym, SUM(pay_amount) AS amt FROM orders WHERE status IN ('已支付', '已发货', '已完成') AND order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01' GROUP BY ym), rf AS (SELECT strftime(refund_time, '%Y-%m') AS ym, SUM(refund_amount) AS amt FROM refunds WHERE status = '已完成' AND refund_time >= TIMESTAMP '2025-01-01' AND refund_time < TIMESTAMP '2026-01-01' GROUP BY ym) SELECT g.ym, ROUND(COALESCE(r.amt, 0) / g.amt * 100, 2) AS refund_rate_pct FROM gmv g LEFT JOIN rf r ON r.ym = g.ym ORDER BY g.ym

-- generated
SELECT
    strftime(r.refund_time, '%Y-%m') AS month,
    SUM(r.refund_amount) AS refund_amount,
    SUM(o.pay_amount) AS sales_amount,
    SUM(r.refund_amount) / NULLIF(SUM(o.pay_amount), 0) * 100 AS refund_rate
FROM refunds r
JOIN orders o ON r.order_id = o.order_id
WHERE r.status = '已完成'
  AND r.refund_time >= TIMESTAMP '2025-01-01'
  AND r.refund_time < TIMESTAMP '2026-01-01'
  AND o.status IN ('已支付', '已发货', '已完成')
GROUP BY strftime(r.refund_time, '%Y-%m')
ORDER BY month
```

- 问题：2025 年客单价最高的前 5 个城市是哪些？（只统计有效订单数不少于 30 单的城市）
- 口径：客单价 = AVG(pay_amount)；过滤条件在 HAVING 而非 WHERE

```sql
-- gold
SELECT u.city, COUNT(*) AS order_cnt, ROUND(AVG(o.pay_amount), 2) AS aov FROM orders o JOIN users u ON u.user_id = o.user_id WHERE o.status IN ('已支付', '已发货', '已完成') AND o.order_time >= TIMESTAMP '2025-01-01' AND o.order_time < TIMESTAMP '2026-01-01' AND u.city IS NOT NULL GROUP BY u.city HAVING COUNT(*) >= 30 ORDER BY aov DESC, u.city LIMIT 5

-- generated
SELECT
    u.city,
    SUM(o.pay_amount) / COUNT(DISTINCT o.order_id) AS avg_order_value
FROM orders o
JOIN users u ON o.user_id = u.user_id
WHERE o.status IN ('已支付', '已发货', '已完成')
  AND o.order_time >= TIMESTAMP '2025-01-01'
  AND o.order_time < TIMESTAMP '2026-01-01'
  AND u.city IS NOT NULL
GROUP BY u.city
HAVING COUNT(DISTINCT o.order_id) >= 30
ORDER BY avg_order_value DESC
LIMIT 5
```

### h02 · hard · 列数不一致

- 问题：2025 年每个品类中销售额最高的商品分别是哪个？
- 口径：组内 Top1，不是全局 Top1

```sql
-- gold
WITH s AS (SELECT p.category, p.product_name, SUM(oi.item_amount) AS amt FROM order_items oi JOIN orders o ON o.order_id = oi.order_id JOIN products p ON p.product_id = oi.product_id WHERE o.status IN ('已支付', '已发货', '已完成') AND o.order_time >= TIMESTAMP '2025-01-01' AND o.order_time < TIMESTAMP '2026-01-01' GROUP BY p.category, p.product_name), r AS (SELECT category, product_name, amt, ROW_NUMBER() OVER (PARTITION BY category ORDER BY amt DESC, product_name) AS rn FROM s) SELECT category, product_name, ROUND(amt, 2) AS amt FROM r WHERE rn = 1 ORDER BY category

-- generated
WITH product_sales AS (
    SELECT
        p.category,
        p.product_id,
        p.product_name,
        SUM(oi.item_amount) AS sales_amount
    FROM order_items oi
    JOIN orders o ON oi.order_id = o.order_id
    JOIN products p ON oi.product_id = p.product_id
    WHERE o.status IN ('已支付', '已发货', '已完成')
      AND o.order_time >= TIMESTAMP '2025-01-01'
      AND o.order_time < TIMESTAMP '2026-01-01'
    GROUP BY p.category, p.product_id, p.product_name
),
ranked AS (
    SELECT
        category,
        product_id,
        product_name,
        sales_amount,
        ROW_NUMBER() OVER (PARTITION BY category ORDER BY sales_amount DESC, product_id) AS rn
    FROM product_sales
)
SELECT
    category,
    product_id,
    product_name,
    sales_amount
FROM ranked
WHERE rn = 1
ORDER BY category
```

### h05 · hard · 结果值不一致

- 问题：2025 年的订单中，用户相邻两次有效下单之间间隔天数的中位数是多少？
- 口径：同用户同日多单的间隔为 0；中位数按天算

```sql
-- gold
WITH o AS (SELECT user_id, order_time, LAG(order_time) OVER (PARTITION BY user_id ORDER BY order_time) AS prev_time FROM orders WHERE status IN ('已支付', '已发货', '已完成')) SELECT ROUND(percentile_cont(0.5) WITHIN GROUP (ORDER BY date_diff('day', prev_time, order_time)), 1) AS median_gap_days FROM o WHERE prev_time IS NOT NULL AND order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01'

-- generated
WITH valid_orders AS (
    SELECT
        user_id,
        order_time,
        LAG(order_time) OVER (PARTITION BY user_id ORDER BY order_time) AS prev_order_time
    FROM orders
    WHERE status IN ('已支付', '已发货', '已完成')
      AND order_time >= TIMESTAMP '2025-01-01'
      AND order_time < TIMESTAMP '2026-01-01'
),
intervals AS (
    SELECT
        user_id,
        date_diff('day', prev_order_time, order_time) AS gap_days
    FROM valid_orders
    WHERE prev_order_time IS NOT NULL
)
SELECT
    median(gap_days) AS median_gap_days
FROM intervals
```

SELECT
    COALESCE(u.member_level, '未知') AS member_level,
