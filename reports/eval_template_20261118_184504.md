```

WITH y AS ( SELECT p.category, EXTRACT(year FROM o.order_time) AS yr, SUM(oi.item_amount) AS amt FROM order_items oi JOIN orders o ON o.order_id = oi.order_id JOIN products p ON p.product_id = oi.product_id WHERE o.status IN ('已支付', '已发货', '已完成') AND (o.order_time >= TIMESTAMP '2024-01-01' AND order_time < TIMESTAMP '2025-01-01' OR o.order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01') GROUP BY 1, 2 ), piv AS ( SELECT category, SUM(CASE WHEN yr = 2024 THEN amt ELSE 0 END) AS amt_2024, SUM(CASE WHEN yr = 2025 THEN amt ELSE 0 END) AS amt_2025 FROM y GROUP BY 1 ) SELECT category, ROUND(amt_2024, 2), ROUND(amt_2025, 2), ROUND((amt_2025 - amt_2024) / amt_2024 * 100, 2) AS yoy_pct FROM piv ORDER BY yoy_pct DESC LIMIT 3
```

- 口径：占比的分母是同一渠道内的退款笔数

```sql
-- gold
SELECT c.channel_name, r.reason, COUNT(*) AS cnt, ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (PARTITION BY c.channel_name), 2) AS pct FROM refunds r JOIN orders o ON o.order_id = r.order_id JOIN channels c ON c.channel_id = o.channel_id WHERE r.status = '已完成' AND r.refund_time >= TIMESTAMP '2025-01-01' AND refund_time < TIMESTAMP '2026-01-01' GROUP BY 1, 2 ORDER BY c.channel_name, cnt DESC
```

- 问题：2025 年各品类销量最高的商品，与前一名之外的「第二名」差距是多少？请给出品类、第一名商品、第一名销售额、第二名销售额和两者差额。
- 口径：组内取前两名再自连接，不是全局取前二

```sql
-- gold
WITH s AS ( SELECT p.category, p.product_name, SUM(oi.item_amount) AS amt FROM order_items oi JOIN orders o ON o.order_id = oi.order_id JOIN products p ON p.product_id = oi.product_id WHERE o.status IN ('已支付', '已发货', '已完成') AND o.order_time >= TIMESTAMP '2025-01-01' AND order_time < TIMESTAMP '2026-01-01' GROUP BY 1, 2 ), r AS ( SELECT category, product_name, amt, ROW_NUMBER() OVER (PARTITION BY category ORDER BY amt DESC, product_name) AS rn FROM s ) SELECT a.category, a.product_name AS top1_product, ROUND(a.amt, 2) AS top1_amt, ROUND(COALESCE(b.amt, 0), 2) AS top2_amt, ROUND(a.amt - COALESCE(b.amt, 0), 2) AS gap FROM r a LEFT JOIN r b ON b.category = a.category AND b.rn = 2 WHERE a.rn = 1 ORDER BY gap DESC
```
